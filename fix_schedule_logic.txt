function getBestShow(
  village: string, afterMin: number, preferredShows: string[], cutoffMin: number,
  fromVillage: string = 'entrance'
): { name: string; time: string; timeMin: number } | null {
  const villageShows = SHOWS[village];
  if (!villageShows) return null;

  const candidates: { name: string; time: string; timeMin: number; score: number }[] = [];

  for (const [name, times] of Object.entries(villageShows)) {
    for (const { time, crowd_density } of times) {
      const timeMin = toMinutes(time);
      if (timeMin < afterMin) continue;
      if (timeMin > cutoffMin) continue;

      const interestMatch  = preferredShows.includes(name) ? 10 : 0;
      const crowdScore     = 8 * (1 - crowd_density);
      const proximityScore = getWalkTime(fromVillage, village) <= ADJACENCY_WALK_MIN ? 4 : 0;
      // Strongly prefer top-of-hour slots (XX:00) over half-hour slots
      const topOfHourBonus = timeMin % 60 === 0 ? 6 : 0;
      const jitter = Math.random() * 1.5;
      candidates.push({
        name, time, timeMin,
        score: interestMatch + crowdScore + proximityScore + topOfHourBonus + jitter,
      });
    }
  }
  if (!candidates.length) return null;

  candidates.sort((a, b) => b.score - a.score || a.timeMin - b.timeMin);
  return weightedRandomSelect(candidates.slice(0, 3));
}

function buildSmartRoute(
  arrMin: number, priority: string[], preferredShows: string[],
  isBusy: boolean, lastShowCutoff: number,
  interests: string[] = [], energy: string = 'medium', group: string = 'solo'
): ScheduleStop[] {
  const stops: ScheduleStop[] = [];
  const usedVillages = new Set<string>();
  let curMin = arrMin;
  let lastVillage: string | null = null;
  let canoeUsed = false;

  // Tracks which stop index is the last show — used to suppress activities after it
  let lastShowIndex = -1;

  function addStop(stop: ScheduleStop | null) {
    if (!stop) return;
    stops.push(stop);
    const dur = stop.type === 'canoe' ? TIMING.CANOE_DURATION_MIN : TIMING.SHOW_DURATION_MIN;
    curMin = stop.timeMin + dur;
    if (stop.type === 'canoe') canoeUsed = true;
    if (stop.type === 'show') {
      lastVillage = stop.village.toLowerCase();
      lastShowIndex = stops.length - 1;
    }
  }

  // Phase 1: Huki
  const hukiMin = toMinutes(SPECIAL_EVENTS.huki.time);
  const hukiEndMin = toMinutes(SPECIAL_EVENTS.huki.endTime);
  if (arrMin <= hukiMin) {
    addStop({
      time: timeToDisplay(SPECIAL_EVENTS.huki.time), timeMin: hukiMin,
      type: 'huki', village: 'All Villages',
      title: SPECIAL_EVENTS.huki.title, desc: SPECIAL_EVENTS.huki.desc,
      flags: ['dont_miss'],
    });
    curMin = hukiEndMin;
  } else if (arrMin > hukiMin && arrMin < hukiEndMin) {
    curMin = hukiEndMin;
  }

  // Phase 2: Build 3 show slots targeting 2:00, 3:00, 4:00 PM
  // Pick the best available village for each target hour from the priority list.
  const targetHours = [
    toMinutes('14:00'),
    toMinutes('15:00'),
    toMinutes('16:00'),
  ].filter(t => t >= curMin && t <= lastShowCutoff);

  // All villages in priority order — no village is excluded or forced
  const allVillages = priority.filter(v => !usedVillages.has(v));

  // For each target hour, find the best village+show combination
  for (const targetMin of targetHours) {
    let bestVillage: string | null = null;
    let bestShow: { name: string; time: string; timeMin: number } | null = null;
    let bestScore = -1;

    for (const v of allVillages) {
      if (usedVillages.has(v)) continue;
      const from = lastVillage ?? 'entrance';
      // afterMin is the walk time from current position to this village
      const afterMin = Math.max(curMin + getWalkTime(from, v), targetMin - 10);
      if (!isTransitionValid(lastVillage, v, canoeUsed, false)) continue;

      const show = getBestShow(v, afterMin, preferredShows, lastShowCutoff, from);
      if (!show) continue;

      // Score: prefer shows closer to the target hour
      const timeDiff = Math.abs(show.timeMin - targetMin);
      const score = 100 - timeDiff + (preferredShows.includes(show.name) ? 20 : 0);
      if (score > bestScore) {
        bestScore = score;
        bestVillage = v;
        bestShow = show;
      }
    }

    if (bestVillage && bestShow) {
      // Insert canoe if needed to reach northern villages
      const needsCanoe = !canoeUsed &&
        ['tahiti', 'hawaii', 'aotearoa'].includes(bestVillage) &&
        ['samoa', 'tonga'].includes(lastVillage ?? '');
      if (needsCanoe) {
        const canoeMin = getNextCanoeTime(curMin, 'north');
        if (canoeMin && canoeMin + TIMING.CANOE_DURATION_MIN <= bestShow.timeMin) {
          addStop(buildCanoeStop(canoeMin, 'north'));
        }
      }

      addStop(buildShowStop(
        bestVillage, bestShow.name, bestShow.time,
        isBusy, usedVillages, interests, energy, group
      ));
    }
  }

  // Phase 3: Insert activities ONLY between Huki end and the 2 PM show
  // Find the 2 PM show stop and insert activities before it at the same village
  const firstShowStop = stops.find(s => s.type === 'show');
  if (firstShowStop && hukiEndMin < firstShowStop.timeMin) {
    const gapStart = hukiEndMin;
    const gapEnd = firstShowStop.timeMin - 5; // 5 min buffer before show
    const actVillage = firstShowStop.village.toLowerCase();
    const acts = assignActivities(actVillage, interests, energy, group);

    let actMin = gapStart;
    const activityInserts: ScheduleStop[] = [];
    for (const actName of acts) {
      if (actMin + 15 > gapEnd) break;
      activityInserts.push({
        time: toDisplayTime(actMin), timeMin: actMin,
        type: 'activity', village: firstShowStop.village,
        title: actName, desc: '', flags: [],
      });
      actMin += 15;
    }

    // Insert activities into stops array right before the first show
    const firstShowIdx = stops.findIndex(s => s.type === 'show');
    stops.splice(firstShowIdx, 0, ...activityInserts);
  }

  return stops.filter(Boolean) as ScheduleStop[];
}
