// ─── Schedule View ────────────────────────────────────────────────────────────

function ScheduleView({ schedule, greeting, isBusy, ticket, onRestart }: {
  schedule: ScheduleResult;
  greeting: string;
  isBusy: boolean;
  ticket: string;
  onRestart: () => void;
}) {
  const viewShotRef = useRef<any>(null);
  const [saving, setSaving] = useState(false);
  const canoeStop = schedule.stops.find(s => s.type === 'canoe');
  const routeLabel = canoeStop ? 'Walk south → Canoe north route' : 'Walking route through the center';

  // ── Reroute toast: track elapsed time on the current stop ──
  const [stops, setStops] = useState<ScheduleStop[]>(schedule.stops);
  useEffect(() => { setStops(schedule.stops); }, [schedule]);

  const [nowMs, setNowMs] = useState(() => Date.now());
  const suppressedUntilRef = useRef(0);
  const elapsedTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);

  useEffect(() => {
    elapsedTimerRef.current = setInterval(() => setNowMs(Date.now()), 30000);
    return () => {
      if (elapsedTimerRef.current) clearInterval(elapsedTimerRef.current);
    };
  }, []);

  const nowDate = new Date(nowMs);
  const nowMin  = nowDate.getHours() * 60 + nowDate.getMinutes();

  let currentIndex = -1;
  for (let i = 0; i < stops.length; i++) {
    if (nowMin >= stops[i].timeMin) currentIndex = i;
  }

  const currentStop      = currentIndex >= 0 ? stops[currentIndex] : null;
  const nextStop         = currentIndex >= 0 ? stops[currentIndex + 1] : undefined;
  const remainingStops   = currentIndex >= 0 ? stops.slice(currentIndex + 1) : [];
  const currentLocation  = currentStop ? currentStop.village : '';
  const elapsedMinutes   = currentStop ? nowMin - currentStop.timeMin : 0;
  const scheduledMinutes = currentStop
    ? (nextStop ? Math.max(15, nextStop.timeMin - currentStop.timeMin) : 30)
    : 0;

  const showRerouteToast = !!currentStop
    && remainingStops.length > 0
    && elapsedMinutes > scheduledMinutes + 5
    && nowMs > suppressedUntilRef.current;

  function handleReroute(newSchedule: ScheduleStop[]) {
    if (currentIndex < 0) return;
    setStops(prev => [...prev.slice(0, currentIndex + 1), ...newSchedule]);
  }

  function handleDismissToast() {
    suppressedUntilRef.current = Date.now() + 15 * 60 * 1000;
    setNowMs(Date.now());
  }

  async function saveSchedule() {
    try {
      setSaving(true);
      const uri = await viewShotRef.current.capture();
      await Sharing.shareAsync(uri, {
        mimeType: 'image/jpeg',
        dialogTitle: 'Save your PCC schedule',
      });
      setSaving(false);
    } catch (e) { setSaving(false); }
  }

  async function shareSchedule() {
    try {
      setSaving(true);
      const uri = await viewShotRef.current.capture();
      await Sharing.shareAsync(uri);
      setSaving(false);
    } catch { setSaving(false); }
  }

  return (
    <View style={{ flex: 1 }}>
    <ScrollView style={{ flex: 1, backgroundColor: D.schedBg }} contentContainerStyle={s.schedContent}>

      {/* Hidden share card */}
      <ViewShot ref={viewShotRef} options={{ format: 'jpg', quality: 0.95 }}
        style={{ position: 'absolute', left: -9999, top: 0 }}>
        <ShareCard schedule={schedule} ticket={ticket} />
      </ViewShot>

      {/* Greeting hero */}
      {greeting ? (
        <View style={s.greetHero}>
          <ImageBackground
            source={IMAGES.tahiti}
            style={StyleSheet.absoluteFill}
            imageStyle={{ borderRadius: 16 }}
            resizeMode="cover">
            <LinearGradient
              colors={['rgba(107,58,42,0.25)', 'rgba(59,16,8,0.92)']}
              style={StyleSheet.absoluteFill}
            />
          </ImageBackground>
          <View style={s.greetContent}>
            <Text style={s.greetEye}>Your guide says</Text>
            <Text style={s.greetText}>{greeting}</Text>
          </View>
        </View>
      ) : null}

      {/* Busy day warning */}
      {isBusy && (
        <View style={s.busyBadge}>
          <Text style={s.busyText}>⚡ Busy day — extra buffer time built into popular shows</Text>
        </View>
      )}

      {/* Route pill */}
      <View style={s.routePill}>
        <Text style={s.routeText}>🗺️ {routeLabel}</Text>
      </View>

      {/* Schedule section */}
      <Text style={s.schedSectionLabel}>Your personalized schedule</Text>

      <View style={s.schedCard}>
        {stops.map((stop, i) => (
          <View key={i}>
            <ScheduleStopRow stop={stop} isLast={i === stops.length - 1} />
            {i < stops.length - 1 && (
              <WalkIndicator
                fromVillage={stop.village}
                toVillage={stops[i + 1].village}
                minutes={getWalkTime(stop.village, stops[i + 1].village)}
              />
            )}
          </View>
        ))}
      </View>

      {/* Upgrade card */}
      {schedule.needsUpgrade && (
        <View style={s.upgradeCard}>
          <Text style={s.upgradeEye}>Upgrade available</Text>
          <Text style={s.upgradeTitle}>Make Tonight{'\n'}Unforgettable</Text>
          <Text style={s.upgradeDesc}>
            Add dinner and Hā: Breath of Life — your night isn't over yet.
          </Text>
          <TouchableOpacity style={s.upgradeBtn}
            onPress={() => Linking.openURL('https://packages.polynesia.com')}>
            <Text style={s.upgradeBtnText}>See Upgrade Options →</Text>
          </TouchableOpacity>
        </View>
      )}

      {/* Free 3 days */}
      {schedule.isLate && (
        <View style={s.free3Card}>
          <Text style={s.free3Title}>🎟 Free Within 3 Days</Text>
          <Text style={s.free3Desc}>
            Arriving late? Your ticket lets you return within 3 days at no charge.
          </Text>
        </View>
      )}

      {/* Save / Share */}
      <TouchableOpacity style={s.saveBtn} onPress={saveSchedule} disabled={saving}>
        <Text style={s.saveBtnText}>{saving ? 'Saving...' : '↓  Save & Share Schedule'}</Text>
      </TouchableOpacity>
      <TouchableOpacity style={s.shareBtn} onPress={shareSchedule} disabled={saving}>
        <Text style={s.shareBtnText}>Share with Friends</Text>
      </TouchableOpacity>

      {/* Restart */}
      <TouchableOpacity style={s.restartBtn} onPress={onRestart}>
        <Text style={s.restartText}>↺  Plan a new day</Text>
      </TouchableOpacity>

    </ScrollView>

    {showRerouteToast && currentStop && (
      <RerouteToast
        currentStop={currentStop}
        remainingStops={remainingStops}
        currentLocation={currentLocation}
        elapsedMinutes={elapsedMinutes}
        scheduledMinutes={scheduledMinutes}
        onReroute={handleReroute}
        onDismiss={handleDismissToast}
      />
    )}
    </View>
  );
}

// ─── Question View ────────────────────────────────────────────────────────────

function QuestionView({ step, answers, onSingle, onMulti, onTicket, onNext, onBack, isComplete, onMagicStart }: {
  step: number;
  answers: PlannerAnswers;
  onSingle: (id: string, v: string) => void;
  onMulti: (id: string, v: string) => void;
  onTicket: (id: string) => void;
  onNext: () => void;
  onBack: () => void;
  isComplete: boolean;
  onMagicStart: () => void;
}) {
  const q = QUESTIONS[step];
  const isLast = step === QUESTIONS.length - 1;
  const progress = ((step + 1) / QUESTIONS.length) * 100;

  return (
    <ScrollView style={{ flex: 1, backgroundColor: D.white }}
      contentContainerStyle={s.qScroll}
      keyboardShouldPersistTaps="handled">

      {step === 0 ? (
        <>
          {/* Question text */}
          <Text style={s.qTitle}>{q.title}</Text>
          <Text style={s.qSub}>{q.sub}</Text>

          {/* Magic Start button */}
          <TouchableOpacity style={s.magicBtn} onPress={onMagicStart} activeOpacity={0.85}>
            <Text style={s.magicBtnText}>Build my day for me</Text>
            <Text style={s.magicBtnSub}>Skip the questions — we'll handle it</Text>

          {/* Step indicator */}
          <View style={s.qStepRow}>
            <Text style={s.qStepNum}>
              {String(step + 1).padStart(2, '0')} / {String(QUESTIONS.length).padStart(2, '0')}
              {'  ·  '}{q.label}
            </Text>

              {QUESTIONS.map((_, i) => (
                <View key={i} style={[s.qDot, i === step && s.qDotActive, i < step && s.qDotDone]} />
              ))}
            </View>
          </View>

          {/* Progress bar */}
          <View style={s.qProgress}>
            <View style={[s.qProgressFill, { width: `${progress}%` as any }]} />
          </View>
        </>
      ) : (
        <>
          {/* Step indicator */}
          <View style={s.qStepRow}>
            <Text style={s.qStepNum}>
              {String(step + 1).padStart(2, '0')} / {String(QUESTIONS.length).padStart(2, '0')}
              {'  ·  '}{q.label}
            </Text>
            <View style={s.qDots}>
              {QUESTIONS.map((_, i) => (
                <View key={i} style={[s.qDot, i === step && s.qDotActive, i < step && s.qDotDone]} />
              ))}
            </View>
          </View>

          {/* Progress bar */}
          <View style={s.qProgress}>
            <View style={[s.qProgressFill, { width: `${progress}%` as any }]} />
          </View>

          {/* Question text */}
          <Text style={s.qTitle}>{q.title}</Text>
          <Text style={s.qSub}>{q.sub}</Text>
        </>
      )}

      {/* Single select — photo grid */}
      {q.type === 'single' && (
        <View style={s.photoGrid}>
          {q.options.map((opt: any) => {
            const sel = (answers as any)[q.id] === opt.value;
            const photo = QUESTION_PHOTOS[opt.value];
            return (
              <TouchableOpacity
                key={opt.value}
                style={[s.photoCard, sel && s.photoCardSel]}
                onPress={() => onSingle(q.id, opt.value)}
                activeOpacity={0.85}>
                {photo && (
                  <ExpoImage
                    source={photo}
                    style={StyleSheet.absoluteFill}
                    contentFit="cover"
                    cachePolicy="memory-disk"
                  />
                )}
                <LinearGradient
                  colors={['transparent', sel ? 'rgba(150,45,10,0.88)' : 'rgba(0,0,0,0.72)']}
                  style={StyleSheet.absoluteFill}
                />
                {sel && (
                  <View style={s.photoCardCheck}>
                    <Text style={{ color: '#fff', fontSize: 12, fontWeight: '700' }}>✓</Text>
                  </View>
                )}
                <View style={s.photoCardLabel}>
                  <Text style={s.photoCardName}>{opt.label}</Text>
                  {opt.sub ? <Text style={s.photoCardSub}>{opt.sub}</Text> : null}
                </View>
              </TouchableOpacity>
            );
          })}
        </View>
      )}

      {/* Connection — photo grid multi-select */}
      {q.type === 'multi' && q.id === 'connection' && (
        <View style={s.photoGrid}>
          {q.options.map((opt: any) => {
            const sel = ((answers as any)[q.id] || []).includes(opt.value);
            const photo = QUESTION_PHOTOS[opt.value];
            return (
              <TouchableOpacity
                key={opt.value}
                style={[s.photoCard, sel && s.photoCardSel]}
                onPress={() => onMulti(q.id, opt.value)}
                activeOpacity={0.85}>
                {photo ? (
                  <ExpoImage
                    source={photo}
                    style={StyleSheet.absoluteFill}
                    contentFit="cover"
                    cachePolicy="memory-disk"
                  />
                ) : (
                  <View style={[StyleSheet.absoluteFill, { backgroundColor: D.brown, borderRadius: 14 }]} />
                )}
                <LinearGradient
                  colors={['transparent', sel ? 'rgba(150,45,10,0.88)' : 'rgba(0,0,0,0.72)']}
                  style={StyleSheet.absoluteFill}
                />
                {sel && (
                  <View style={s.photoCardCheck}>
                    <Text style={{ color: '#fff', fontSize: 12, fontWeight: '700' }}>✓</Text>
                  </View>
                )}
                <View style={s.photoCardLabel}>
                  <Text style={s.photoCardName}>{opt.label}</Text>
                  {opt.sub ? <Text style={s.photoCardSub}>{opt.sub}</Text> : null}
                </View>
              </TouchableOpacity>
            );
          })}
        </View>
      )}

      {/* Vibe — clean chip multi-select */}
      {q.type === 'multi' && q.id !== 'connection' && (
        <View style={s.chipGrid}>
          {q.options.map((opt: any) => {
            const sel = ((answers as any)[q.id] || []).includes(opt.value);
            return (
              <TouchableOpacity
                key={opt.value}
                style={[s.chip, sel && s.chipSel]}
                onPress={() => onMulti(q.id, opt.value)}
                activeOpacity={0.75}>
                <Text style={[s.chipLabel, sel && s.chipLabelSel]}>{opt.label}</Text>
                <Text style={s.chipSub}>{opt.sub}</Text>
              </TouchableOpacity>
            );
          })}
        </View>
      )}

      {/* Arrival time */}
      {q.type === 'select' && (
        <>
          <View style={s.timeGrid}>
            {QUESTIONS[step].options.map((opt: any) => {
              const sel = (answers as any)[q.id] === opt;
              return (
                <TouchableOpacity
                  key={opt}
                  style={[s.timeChip, sel && s.timeChipSel]}
                  onPress={() => onSingle(q.id, opt)}
                  activeOpacity={0.75}>
                  <Text style={[s.timeChipText, sel && s.timeChipTextSel]}>{opt}</Text>
                </TouchableOpacity>
              );
            })}
          </View>
          <Text style={s.arrivalNote}>PCC open 12:15 PM – 5:30 PM · Closed Sundays</Text>
        </>
      )}

      {/* Ticket selection */}
      {q.type === 'ticket' && (
        <View style={s.ticketList}>
          {Object.values(TICKETS).map((ticket) => {
            const sel = (answers as any).ticket === ticket.id;
            const barColor =
              ticket.id === 'islands'    ? '#B08070' :
              ticket.id === 'islands-ha' ? '#0277BD' :
              ticket.id === 'gateway'    ? D.orange  : D.brown;
            return (
              <View key={ticket.id}>
                {ticket.divider && (
                  <View style={s.ticketDivider}>
                    <View style={s.ticketDividerLine} />
                    <Text style={s.ticketDividerText}>Declined guided tour</Text>
                    <View style={s.ticketDividerLine} />
                  </View>
                )}
                <TouchableOpacity
                  style={[s.ticketCard, sel && s.ticketCardSel]}
                  onPress={() => onTicket(ticket.id)}
                  activeOpacity={0.8}>
                  <View style={[s.ticketBar, { backgroundColor: barColor }]} />
                  <View style={s.ticketBody}>
                    {ticket.badge === 'popular' && (
                      <View style={s.ticketBadgeWrap}>
                        <Text style={[s.ticketBadge, { color: '#9A5B00', backgroundColor: '#FFF3EE' }]}>Most Popular</Text>
                      </View>
                    )}
                    {ticket.badge === 'show' && (
                      <View style={s.ticketBadgeWrap}>
                        <Text style={[s.ticketBadge, { color: '#0277BD', backgroundColor: '#E3F2FD' }]}>Includes Hā Show</Text>
                      </View>
                    )}
                    {ticket.badge === 'self' && (
                      <View style={s.ticketBadgeWrap}>
                        <Text style={[s.ticketBadge, { color: D.brown, backgroundColor: D.cream }]}>Self-guided</Text>
                      </View>
                    )}
                    <Text style={s.ticketName}>{ticket.name}</Text>
                    <Text style={s.ticketDesc}>{ticket.desc}</Text>
                  </View>
                </TouchableOpacity>
              </View>
            );
          })}
        </View>
      )}

      {/* Nav row */}
      <View style={s.navRow}>
        {step > 0 && (
          <TouchableOpacity style={s.backBtn} onPress={onBack}>
            <Text style={s.backBtnIcon}>←</Text>
          </TouchableOpacity>
        )}
        {(q.type !== 'single' || (answers as any)[q.id]) && (
          <TouchableOpacity
            style={[s.mainBtn, !isComplete && s.mainBtnDisabled]}
            onPress={onNext}
            disabled={!isComplete}
            activeOpacity={0.85}>
            <Text style={s.mainBtnText}>
              {isLast ? 'Build My Schedule' : 'Continue'}
            </Text>
            <View style={s.mainBtnArrow}>
              <Text style={{ color: '#fff', fontSize: 16 }}>→</Text>
            </View>
          </TouchableOpacity>
        )}
      </View>

      {/* Footer watermark */}
      <Text style={s.qFooter}>POLYNESIAN CULTURAL CENTER</Text>

    </ScrollView>
  );
}

// ─── Chat Bubble ──────────────────────────────────────────────────────────────

function ChatBubble({ msg }: { msg: ChatMessage }) {
  const jsonMatch = msg.content.match(/SCHEDULE_JSON_START([\s\S]*?)SCHEDULE_JSON_END/);
  const displayText = msg.content.replace(/SCHEDULE_JSON_START[\s\S]*?SCHEDULE_JSON_END/g, '').trim();

  if (msg.role === 'user') {
    return (
      <View style={[s.msgRow, { justifyContent: 'flex-end' }]}>
        <View style={s.userBubble}>
          <Text style={{ color: '#fff', fontSize: 13, lineHeight: 20 }}>{displayText}</Text>
        </View>
      </View>
    );
  }
  return (
    <View style={s.msgRow}>
      <View style={s.msgAvatar}>
        <Text style={{ fontSize: 11 }}>📖</Text>
      </View>
      <View style={{ flex: 1, gap: 4 }}>
        {displayText ? (
          <View style={s.aiBubble}>
            <Text style={{ fontSize: 13, lineHeight: 20, color: D.dark }}>{displayText}</Text>
          </View>
        ) : null}
        {jsonMatch ? <ChatScheduleCard jsonStr={jsonMatch[1].trim()} /> : null}
      </View>
    </View>
  );
}

function ChatScheduleCard({ jsonStr }: { jsonStr: string }) {
  let data: ScheduleJsonData;
  try { data = JSON.parse(jsonStr); } catch { return null; }
  return (
    <View style={s.miniCard}>
      <View style={s.miniCardHeader}>
        <Text style={{ color: '#fff', fontSize: 12, fontWeight: '600' }}>{data.greeting || 'Your Schedule'}</Text>
      </View>
      {(data.items || []).map((item, i) => (
        <View key={i} style={[s.miniCardItem, i === (data.items.length - 1) && { borderBottomWidth: 0 }]}>
          <Text style={s.miniTime}>{item.time}</Text>
          <View style={{ flex: 1 }}>
            <Text style={s.miniVillage}>{item.village}</Text>
            <Text style={s.miniTitle}>{item.title}</Text>
          </View>
        </View>
      ))}
    </View>
  );
}

// ─── Main Screen ──────────────────────────────────────────────────────────────

export default function PlannerScreen() {
  const [step, setStep]         = useState(0);
  const [answers, setAnswers]   = useState<PlannerAnswers>({});
  const [loading, setLoading]   = useState(false);
  const [schedule, setSchedule] = useState<ScheduleResult | null>(null);
  const [greeting, setGreeting] = useState('');
  const [chatOpen, setChatOpen] = useState(false);
  const [chatInit, setChatInit] = useState(false);
  const [chatMsgs, setChatMsgs] = useState<ChatMessage[]>([]);
  const [chatInput, setChatInput] = useState('');
  const [chatBusy, setChatBusy]   = useState(false);
  const [quickReplies, setQuickReplies] = useState<string[]>([]);

  const drawerAnim   = useRef(new Animated.Value(0)).current;
  const chatScrollRef = useRef<ScrollView>(null);

  const today   = new Date();
  const dayName = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'][today.getDay()];
  const isBusy  = [1, 4, 5].includes(today.getDay());
  const progress = Math.round((step / QUESTIONS.length) * 100);

  function selectSingle(id: string, value: string) {
    setAnswers(a => ({ ...a, [id]: value }));
  }

  function toggleMulti(id: string, value: string) {
    setAnswers(a => {
      const cur: string[] = (a as any)[id] || [];
      let next: string[];
      if (value === 'none') {
        next = ['none'];
      } else {
        const withoutNone = cur.filter(v => v !== 'none');
        next = withoutNone.includes(value)
          ? withoutNone.filter(v => v !== value)
          : [...withoutNone, value];
      }
      return { ...a, [id]: next };
    });
  }

  function selectTicket(id: string) {
    setAnswers(a => ({ ...a, ticket: id }));
  }

  function isComplete() {
    const q = QUESTIONS[step];
    switch (q.type) {
      case 'single': return !!(answers as any)[q.id];
      case 'multi':  return ((answers as any)[q.id] || []).length > 0;
      case 'select': return !!(answers as any)[q.id];
      case 'ticket': return !!answers.ticket;
      default:       return true;
    }
  }

  function advance() {
    if (!isComplete()) return;
    if (step < QUESTIONS.length - 1) {
      setStep(s => s + 1);
    } else {
      buildPlan();
    }
  }

  function back() {
    if (step > 0) setStep(s => s - 1);
  }

  function restart() {
    setStep(0);
    setAnswers({});
    setSchedule(null);
    setGreeting('');
  }

  async function buildPlan() {
    setLoading(true);
    const {
      group = 'solo', energy = 'medium', vibe = [],
      connection = [], arrival = '1:00 PM', ticket = 'islands'
    } = answers;
    const arrivalHHMM = ARRIVAL_TIME_MAP[arrival] || '13:00';
    const result = buildSchedule({
      arrival: arrivalHHMM, group, interests: vibe,
      connection, ticket, isBusy, energy,
    });
    const gr = await fetchGreeting(group, vibe, connection);
    setGreeting(gr || getDefaultGreeting());
    setSchedule(result);
    setLoading(false);
  }

  function magicStart() {
    const now = new Date();
    const arrivalHHMM = `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

    // Standard family persona for the Magic Start shortcut
    const group      = 'family_kids';
    const energy     = 'medium';
    const connection = ['samoa', 'hawaii'];
    const interests: string[] = [];
    const ticket     = 'islands-ha';

    setStep(QUESTIONS.length - 1);
    setAnswers({ group, energy, vibe: interests, connection, ticket });
    setLoading(true);

    const result = buildSchedule({
      arrival: arrivalHHMM, group, interests,
      connection, ticket, isBusy, energy,
    });
    setSchedule(result);

    fetchGreeting(group, interests, connection).then(gr => {
      setGreeting(gr || getDefaultGreeting());
      setLoading(false);
    });
  }

  function openChat() {
    setChatOpen(true);
    Animated.spring(drawerAnim, { toValue: 1, useNativeDriver: true, tension: 65, friction: 11 }).start();
    if (!chatInit) {
      setChatInit(true);
      const opening = isBusy
        ? `Aloha! 🌺 It's a busy ${dayName} — I'll help you beat the crowds.`
        : `Aloha! 🌺 Welcome to PCC! I'm here to help plan your perfect day.`;
      setChatMsgs([{ role: 'assistant', content: opening }]);
      setQuickReplies(['Just me', 'My family', "Couple's trip", 'Group of friends']);
    }
  }

  function closeChat() {
    Animated.timing(drawerAnim, { toValue: 0, duration: 280, useNativeDriver: true })
      .start(() => setChatOpen(false));
  }

  async function sendChat(text: string) {
    if (!text.trim() || chatBusy) return;
    const userMsg: ChatMessage = { role: 'user', content: text.trim() };
    const newHistory = [...chatMsgs, userMsg];
    setChatMsgs(newHistory);
    setChatInput('');
    setQuickReplies([]);
    setChatBusy(true);
    const systemPrompt = buildChatSystemPrompt(isBusy, dayName);
    const reply = await sendChatMessage(
      newHistory.map(m => ({ role: m.role, content: m.content })),
      systemPrompt
    );
    setChatMsgs(h => [...h, { role: 'assistant', content: reply }]);
    setChatBusy(false);
    const lower = reply.toLowerCase();
    if (lower.includes('ticket') || lower.includes('package')) {
      setQuickReplies(['Islands of Polynesia', 'Islands + Hā Show', 'Gateway Buffet', "Self-Guided Ali'i Lū'au"]);
    } else if (lower.includes('arriv') || lower.includes('time')) {
      setQuickReplies(['12:30 PM', '1:00 PM', '2:00 PM', '3:00 PM', '4:00 PM']);
    } else {
      setQuickReplies([]);
    }
  }

  useEffect(() => {
    if (chatScrollRef.current) {
      setTimeout(() => chatScrollRef.current?.scrollToEnd({ animated: true }), 100);
    }
  }, [chatMsgs]);

  const drawerTranslate = drawerAnim.interpolate({ inputRange: [0,1], outputRange: [600,0] });

  return (
    <View style={s.root}>
      <SafeAreaView style={s.safeArea} edges={['top']}>

        {/* Nav header */}
        <View style={s.navBar}>
          <View>
            <Text style={s.navEye}>Polynesian Cultural Center</Text>
            <Text style={s.navTitle}>Plan My Day</Text>
          </View>
          <TouchableOpacity style={s.navSettings}>
            <Text style={{ color: 'rgba(255,255,255,0.7)', fontSize: 18 }}>⚙</Text>
          </TouchableOpacity>
        </View>

        {/* Progress bar */}
        <View style={s.progressBar}>
          <View style={[s.progressFill, {
            width: `${loading || schedule ? 100 : progress}%` as any
          }]} />
        </View>

        {/* Content */}
        {loading ? (
          <LoadingScreen />
        ) : schedule ? (
          <ScheduleView
            schedule={schedule}
            greeting={greeting}
            isBusy={isBusy}
            ticket={answers.ticket || 'islands'}
            onRestart={restart}
          />
        ) : (
          <QuestionView
            step={step}
            answers={answers}
            onSingle={selectSingle}
            onMulti={toggleMulti}
            onTicket={selectTicket}
            onNext={advance}
            onBack={back}
            isComplete={isComplete()}
            onMagicStart={magicStart}
          />
        )}
      </SafeAreaView>

      {/* Chat FAB */}
      <TouchableOpacity style={s.fab} onPress={chatOpen ? closeChat : openChat} activeOpacity={0.85}>
        <Text style={s.fabIcon}>{chatOpen ? '✕' : '💬'}</Text>
      </TouchableOpacity>

      {/* Chat Drawer */}
      {(chatOpen || chatMsgs.length > 0) && (
        <>
          {chatOpen && <Pressable style={s.overlay} onPress={closeChat} />}
          <Animated.View style={[s.drawer, { transform: [{ translateY: drawerTranslate }] }]}>
            <View style={s.drawerHandle} />
            <View style={s.drawerHeader}>
              <View style={s.drawerAvatar}>
                <Text style={{ color: '#fff', fontSize: 14 }}>📖</Text>
              </View>
              <View style={{ flex: 1 }}>
                <Text style={s.drawerName}>Your PCC Guide</Text>
                <Text style={s.drawerStatus}>● Online now</Text>
              </View>
              <TouchableOpacity onPress={closeChat} style={{ padding: 6 }}>
                <Text style={{ fontSize: 18, color: '#aaa' }}>✕</Text>
              </TouchableOpacity>
            </View>
            <ScrollView ref={chatScrollRef} style={s.chatArea}
              contentContainerStyle={{ gap: 8, paddingBottom: 8 }}
              keyboardShouldPersistTaps="handled">
              {chatMsgs.map((msg, i) => <ChatBubble key={i} msg={msg} />)}
              {chatBusy && (
                <View style={s.msgRow}>
                  <View style={s.msgAvatar}><Text style={{ fontSize: 11 }}>📖</Text></View>
                  <View style={s.typingBubble}><ActivityIndicator size="small" color="#bbb" /></View>
                </View>
              )}
            </ScrollView>
            {quickReplies.length > 0 && (
              <ScrollView horizontal style={s.quickReplies} showsHorizontalScrollIndicator={false}
                contentContainerStyle={{ gap: 5, paddingHorizontal: 12, paddingVertical: 6 }}>
                {quickReplies.map((r, i) => (
                  <TouchableOpacity key={i} style={s.qrBtn} onPress={() => sendChat(r)}>
                    <Text style={s.qrText}>{r}</Text>
                  </TouchableOpacity>
                ))}
              </ScrollView>
            )}
            <KeyboardAvoidingView behavior={Platform.OS === 'ios' ? 'padding' : undefined}>
              <View style={s.inputArea}>
                <View style={s.inputWrap}>
                  <TextInput style={s.input} value={chatInput} onChangeText={setChatInput}
                    placeholder="Ask your guide anything..."
                    placeholderTextColor="#bbb" multiline
                    onSubmitEditing={() => sendChat(chatInput)} blurOnSubmit />
                </View>
                <TouchableOpacity
                  style={[s.sendBtn, (!chatInput.trim() || chatBusy) && s.sendBtnDisabled]}
                  onPress={() => sendChat(chatInput)}
                  disabled={!chatInput.trim() || chatBusy}>
                  <Text style={{ color: '#fff', fontSize: 16 }}>↑</Text>
                </TouchableOpacity>
              </View>
            </KeyboardAvoidingView>
          </Animated.View>
        </>
      )}
    </View>
  );
}
